cmake_minimum_required(VERSION 3.20)
project(trivial_simd LANGUAGES C)

function(configure_fma target)
  target_sources(${target} PRIVATE native/fma.c)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$"
     AND NOT CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    target_sources(${target} PRIVATE native/fma-x86.c)
    target_compile_definitions(${target} PRIVATE TS_HAVE_HARDWARE_FMA)
    if(MSVC)
      set_source_files_properties(native/fma-x86.c PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
      set_source_files_properties(native/fma-x86.c PROPERTIES COMPILE_OPTIONS "-mavx;-mfma")
    endif()
  endif()
endfunction()

function(configure_native_library target)
  set_target_properties(${target} PROPERTIES
    C_STANDARD 11 C_STANDARD_REQUIRED YES WINDOWS_EXPORT_ALL_SYMBOLS YES)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -O3 -ffp-contract=off)
    target_link_libraries(${target} PRIVATE m)
  endif()
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${target} PROPERTIES
      LIBRARY_OUTPUT_DIRECTORY_${config} "${PROJECT_BINARY_DIR}"
      RUNTIME_OUTPUT_DIRECTORY_${config} "${PROJECT_BINARY_DIR}")
  endforeach()
endfunction()

add_library(trivial_simd SHARED native/trivial_simd.c native/extended.c native/complex.c native/blas.c)
configure_native_library(trivial_simd)
configure_fma(trivial_simd)

option(BUILD_KERNEL_PROFILE "Build the optional kernel profiling library" OFF)
set(KERNEL_PROFILE_BASELINE_SOURCE "" CACHE FILEPATH "Native VM source for profiling a baseline")
if(BUILD_KERNEL_PROFILE)
  set(profile_targets native_kernel_profile)
  if(KERNEL_PROFILE_BASELINE_SOURCE)
    list(APPEND profile_targets native_kernel_profile_baseline)
  endif()
  foreach(target IN LISTS profile_targets)
    add_library(${target} SHARED tests/kernel-profile.c)
    configure_native_library(${target})
    configure_fma(${target})
  endforeach()
  if(KERNEL_PROFILE_BASELINE_SOURCE)
    target_compile_definitions(native_kernel_profile_baseline PRIVATE
      TS_PROFILE_BASELINE TS_PROFILE_VM="${KERNEL_PROFILE_BASELINE_SOURCE}")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(target native_encoding_test native_encoding_scalar_test)
    add_executable(${target} tests/native-encodings.c)
    configure_native_library(${target})
  endforeach()
  target_compile_definitions(native_encoding_scalar_test PRIVATE TS_SCALAR)
  add_test(NAME native-encodings COMMAND native_encoding_test)
  add_test(NAME native-encodings-scalar COMMAND native_encoding_scalar_test)
  foreach(target native_kernel_test native_kernel_scalar_test native_kernel_software_fma_test)
    add_executable(${target} tests/native-kernel.c)
    configure_fma(${target})
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    if(MSVC)
      target_compile_options(${target} PRIVATE /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -ffp-contract=off)
      target_link_libraries(${target} PRIVATE m)
    endif()
  endforeach()
  target_compile_definitions(native_kernel_scalar_test PRIVATE TS_SCALAR)
  target_compile_definitions(native_kernel_software_fma_test PRIVATE TS_FORCE_SOFTWARE_FMA)
  add_test(NAME native-kernel COMMAND native_kernel_test)
  add_test(NAME native-kernel-scalar COMMAND native_kernel_scalar_test)
  add_test(NAME native-kernel-software-fma COMMAND native_kernel_software_fma_test)
endif()
